## 🎓 GETTING STARTED

### Installation
```bash
npm install @lp-labs/ui-components@4.0.0
```

### Setup
```tsx
import { ThemeProvider } from '@lp-labs/ui-components/theme';
import { ToastProvider } from '@lp-labs/ui-components/components';
import '@lp-labs/ui-components/styles';

function App() {
  return (
    <ThemeProvider>
      <ToastProvider>
        <YourApp />
      </ToastProvider>
    </ThemeProvider>
  );
}
```
