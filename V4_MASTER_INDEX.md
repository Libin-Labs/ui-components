## 🚀 Getting Started

### Installation
```bash
npm install @lp-labs/ui-components@4.0.0
```

### Import Styles
```tsx
import '@lp-labs/ui-components/styles';
```

### Import Components
```tsx
import { 
  ButtonPrimary, 
  CardBasic, 
  InputEmail 
} from '@lp-labs/ui-components';
```

### Use in Application
```tsx
export default function App() {
  return (
    <CardBasic shadow="lg">
      <h1>Welcome to v4.0.0</h1>
      <InputEmail label="Email" />
      <ButtonPrimary>Submit</ButtonPrimary>
    </CardBasic>
  );
}
```
